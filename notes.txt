To check other user details:
SELECT au.username, au.email, up.is_online, up.age, up.nationality, up.bio
FROM auth_user au
JOIN user_userprofile up ON au.id = up.user_id;

To delete users:
DELETE FROM user_userprofile WHERE user_id IN (SELECT id FROM auth_user);
DELETE FROM auth_user;

class Tournament(models.Model):

	id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
	player = object
	max_players = models.IntegerField(default=4)
	is_finished = models.BooleanField(default=False)
	is_ongoing = models.BooleanField(default=False)
	winner = models.ForeignKey(User, on_delete=models.SET_NULL, null=True, blank=True, related_name="won_matches")
	looser = models.ForeignKey(User, on_delete=models.SET_NULL, null=True, blank=True, related_name="lost_matches")
	created_at = models.DateTimeField(auto_now_add=True)
